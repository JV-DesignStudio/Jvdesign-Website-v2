Java 2D Game Dev Cheat Sheet

A one-page printable reference for JFrame setup, game loop, Graphics2D, input, collision, sound and common Java game patterns. Pin it next to your monitor!

⚙️ JFrame Setup

import javax.swing.*;
import java.awt.*;

JFrame frame = new JFrame("My Game");
GamePanel panel = new GamePanel();
panel.setPreferredSize(
    new Dimension(800, 600));

frame.add(panel);
frame.pack();
frame.setDefaultCloseOperation(
    JFrame.EXIT_ON_CLOSE);
frame.setResizable(false);
frame.setVisible(true);

bgr.drawImage(scaled, 0, 0, null);
bgr.dispose();

    new KeyListener() {

      KeyEvent e) {

        dx = 5;

      KeyEvent e) {

📐 Vector2 Helper

class Vector2 {
    double x, y;

    Vector2(double x, double y) {
        this.x = x;
        this.y = y;
    }

    Vector2 add(Vector2 v) {
        return new Vector2(x+v.x, y+v.y);
    }

    Vector2 multiply(double s) {
        return new Vector2(x*s, y*s);
    }

    double magnitude() {
        return Math.sqrt(x*x + y*y);
    }

    Vector2 normalize() {
        double m = magnitude();
        return new Vector2(x/m, y/m);
    }
}

🛠️ Common Patterns

── GAME LOOP WITH DELTA TIME ──
long lastTime = System.nanoTime();
double fps = 60.0;
double ns = 1000000000.0 / fps;

while (true) {
    long now = System.nanoTime();
    double delta = (now - lastTime) / ns;
    update(delta);
    repaint();
    lastTime = now;
}

── ENTITY SYSTEM ──
class Entity {
    double x, y, w, h, dx, dy;
    boolean alive = true;
    abstract void update(double dt);
    abstract void draw(Graphics2D g);
}

class EntityManager {
    List<Entity> entities = new ArrayList<>();
    void update(double dt) {
        entities.forEach(e -> e.update(dt));
        entities.removeIf(e -> !e.alive);
    }
    void draw(Graphics2D g) {
        entities.forEach(e -> e.draw(g));
    }
}

── SPRITE SHEET ANIMATION ──
int frame = 0, frameDelay = 10, tick = 0;
int frameW = img.getWidth() / cols;
int frameH = img.getHeight() / rows;

// in update():
tick++;
if (tick >= frameDelay) {
    tick = 0;
    frame = (frame + 1) % totalFrames;
}

// in draw():
int sx = (frame % cols) * frameW;
int sy = (frame / cols) * frameH;
g2d.drawImage(img, x, y, x+frameW,
    y+frameH, sx, sy, sx+frameW,
    sy+frameH, null);

── TILE MAP ──
int tileSize = 32;
for (int row = 0; row < map.length; row++) {
    for (int col = 0; col < map[0].length; col++) {
        int tile = map[row][col];
        g2d.drawImage(tileset[tile],
            col*tileSize, row*tileSize, null);
