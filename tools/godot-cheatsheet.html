📡 Signals

# Define custom signal
signal health_changed(new_hp)

# Emit signal
health_changed.emit(hp)

# Connect in _ready
health_changed.connect(_on_health_changed)

# Connect node signal via editor
body_entered.connect(_on_body_entered)

# Disconnect
health_changed.disconnect(_on_health_changed)

# @onready — set after _ready

🛠️ Common Patterns

── PLAYER MOVEMENT (TOP-DOWN) ──
@export var speed = 200.0

func _physics_process(delta):
    var input = Input.get_vector("ui_left", "ui_right",
        "ui_up", "ui_down")
    velocity = input * speed
    move_and_slide()

── ENEMY AI (CHASE) ──
@export var speed = 100.0
@onready var player = get_tree().current_scene.get_node("Player")

func _physics_process(delta):
    var dir = (player.global_position - global_position).normalized()
    velocity = dir * speed
    move_and_slide()

── HEALTH SYSTEM ──
signal died
var hp = 100

func take_damage(amount):
    hp -= amount
    if hp <= 0:
        died.emit()
        queue_free()

── SCENE TRANSITIONS ──
get_tree().change_scene_to_file("res://level2.tscn")
get_tree().reload_current_scene()

var save_file = FileAccess.open("user://save.json", FileAccess.WRITE)
save_file.store_string(JSON.stringify(data))
