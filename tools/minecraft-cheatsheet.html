Minecraft Modding Cheat Sheet

A one-page printable reference for Minecraft Forge/Fabric modding covering mod structure, registry, items, blocks, events, GUI, rendering, networking and data storage.

✨️ Mod Structure

// Main mod class (Forge)
@Mod("mymod")
public class MyMod {
    public MyMod() {
        FMLJavaModLoadingContext
            .get().getModEventBus()
            .register(this);
    }
}

// Event bus for mod lifecycle
IEventBus bus = FMLJavaModLoadingContext
    .get().getModEventBus();

// Register items, blocks, etc.
DeferredRegister<Item> ITEMS =
    DeferredRegister.create(
        ForgeRegistries.ITEMS, "mymod");

📦 Registry

// DeferredRegister pattern
DeferredRegister<Item> ITEMS =
    DeferredRegister.create(
        ForgeRegistries.ITEMS, "mymod");

// Register an item
RegistryObject<Item> MY_ITEM =
    ITEMS.register("my_item",
        () -> new Item(
            new Item.Properties()
                .tab(CreativeModeTab.TAB_MISC)));

// Register a block
DeferredRegister<Block> BLOCKS =
    DeferredRegister.create(
        ForgeRegistries.BLOCKS, "mymod");

RegistryObject<Block> MY_BLOCK =
    BLOCKS.register("my_block",
        () -> new Block(Block.Behaviour
            .of()
            .strength(2.f, 6.f)
            .requiresCorrectToolForDrops()));

🎨 Items

ItemBase class for all items
Item.PropertiesBuilder for item configuration
ItemStackRepresents a stack of items
ItemHandlerHelperUtility for item manipulation
CreativeModeTabTab in creative inventory
public class SwordItem extends SwordItem {
    public SwordItem() {
        super(5, -2.4f,
            new Item.Properties()
                .tab(CreativeModeTab.TAB_COMBAT)
                .durability(100));
    }

    @Override
    public InteractionResultHolder<ItemStack>
    use(Level level, Player player,
        InteractionHand hand) {
        // item right-click logic
        return InteractionResultHolder
            .success(player.getItemInHand(hand));
    }
}

🧱 Blocks

// Block class
public class MyBlock extends Block {
    public MyBlock(Properties props) {
        super(props);
    }
}

// Key types
BlockBehaviour.Properties  - block settings
BlockState                - state of block in world
BlockPos                  - 3D position (x, y, z)
Level                     - the world dimension
BlockEntity               - tile entity / block entity

// Block interaction
@Override
public InteractionResult use(
    BlockState state, Level level,
    BlockPos pos, Player player,
    BlockHitResult hit) {
    // right-click block logic
    return InteractionResult.sidedSuccess(
        level.isClientSide);
}

🌐 Networking

// Channel setup
SimpleChannel CHANNEL = NetworkRegistry
    .newSimpleChannel(
        new ResourceLocation("mymod", "main"),
        () -> "1", "1"::equals,
        "1"::equals);

// Register message
CHANNEL.registerMessage(0,
    MyPacket.class,
    MyPacket::encode,
    MyPacket::decode,
    MyPacket::handle);

// Send to server
CHANNEL.sendToServer(
    new MyPacket(data));

// Send to client
CHANNEL.send(
    PacketDistributor.PLAYER
        .with(() -> player),
    new MyPacket(data));

💾 Data (NBT / CompoundTag)

// CompoundTag (NBT data)
CompoundTag tag = new CompoundTag();
tag.putString("Name", "Steve");
tag.putInt("Level", 42);
tag.putBoolean("Active", true);

// Read data
String name = tag.getString("Name");
int level = tag.getInt("Level");

// ListTag (arrays)
ListTag list = new ListTag();
list.add(StringTag.valueOf("hello"));

// Save to block entity
@Override
public CompoundTag getUpdateTag() {
    return save(new CompoundTag());
}

// Load from block entity
@Override
public void load(CompoundTag tag) {
    super.load(tag);
    name = tag.getString("Name");
}

🛠️ Common Patterns

── CUSTOM ITEM ──
public class FoodItem extends Item {
    public FoodItem() {
        super(new Item.Properties()
            .tab(CreativeModeTab.TAB_FOOD)
            .food(Foods.GOLDEN_APPLE)
            .stacksTo(16));
    }
}

── CUSTOM BLOCK ──
public class LampBlock extends Block {
    public LampBlock() {
        super(Block.Behaviour.of()
            .strength(1.f)
            .lightLevel(state -> state.getValue(LIT) ? 15 : 0));
    }
}

── CUSTOM MOB ──
public class GremlinEntity extends Monster {
    protected void registerGoals() {
        goalSelector.addGoal(2,
            new MeleeAttackGoal(this, 1.2f, false));
        goalSelector.addGoal(7,
            new FloatGoal(this));
    }
}

── GUI / SCREEN ──
// Server-side: open screen
player.openMenu(provider);
// Client-side: register screen
MenuScreens.register(
    MyMenu.TYPE, MyScreen::new);

── WORLD GENERATION ──
// ore gen in datapack or code
