💬 Messages

-- send a message to another component
msg.post("/enemy_factory", "spawn", {
    position = go.get_position(),
    type = "slime"
})

-- broadcast to all components in this GO
msg.post("#", "take_damage", {amount = 10})

-- common message targets
"#"        = this game object
"#sprite"  = the sprite component
"#script"  = the script component
"#camera"  = the camera component

-- receive in on_message:
function on_message(self, id, msg)
    if id == hash("take_damage") then
        self.hp -= msg.amount
    end
end

    local from = go.get_world_position()

    local result = physics.raycast(from, to, {hash("ground")})
    if result then

🏭 Collections & Factories

-- spawn from a factory component
function spawn_enemy(self, pos)
    local props = {hp = 30, speed = 5}
    collectionfactory.create(
        "/enemy_factory#factory",
        pos,
        vmath.quat(),
        props,
        nil,
        function(self, id)
            msg.post(id, "init_enemy",
                {hp = props.hp})
        end
    )
end

🛠️ Common Patterns

── PLAYER MOVEMENT (TOP-DOWN) ──
function update(self, dt)
    local move = vmath.vector3(0)
    if action.pressed then
        if action_id == hash("up") then move.y = 1 end
        if action_id == hash("down") then move.y = -1 end
        if action_id == hash("left") then move.x = -1 end
        if action_id == hash("right") then move.x = 1 end
    end
    if vmath.length(move) > 0 then
        move = vmath.normalize(move)
        go.set_position(go.get_position() + move * self.speed * dt)
    end
end

── HEALTH SYSTEM ──
function on_message(self, id, msg)
    if id == hash("take_damage") then
        self.hp = math.max(0, self.hp - msg.amount)
        if self.hp <= 0 then
            msg.post("@system:", "shutdown")
        end
    end
end

── ENEMY SPAWN TIMER ──
function update(self, dt)
    self.timer = self.timer - dt
    if self.timer <= 0 then
        spawn_enemy(self, go.get_position())
        self.timer = 3  -- every 3 seconds
    end
end
