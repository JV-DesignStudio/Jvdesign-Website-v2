C++ SFML Game Dev Cheat Sheet

A one-page printable reference for SFML window setup, drawing, input, math, animation, collision, audio and common C++ game patterns. Pin it next to your monitor!

⚙️ Window Setup

#include <SFML/Graphics.hpp>

sf::RenderWindow window(
    sf::VideoMode(800, 600),
    "My Game");
window.setFramerateLimit(60);

sf::Clock clock;

while (window.isOpen()) {
    sf::Event event;
    while (window.pollEvent(event)) {
        if (event.type ==
            sf::Event::Closed)
            window.close();
    }
    sf::Time dt = clock.restart();
    // update & draw
    window.display();
}

✏️ Drawing

# rectangle
sf::RectangleShape rect(
    sf::Vector2f(100.f, 50.f));
rect.setPosition(200.f, 150.f);
rect.setFillColor(sf::Color::Blue);
window.draw(rect);

# circle
sf::CircleShape circle(40.f);
circle.setPosition(300.f, 200.f);
window.draw(circle);

# sprite
sf::Texture tex;
tex.loadFromFile("player.png");
sf::Sprite sprite(tex);
window.draw(sprite);

# text
sf::Font font;
font.loadFromFile("arial.ttf");
sf::Text text("Hello", font, 24);
window.draw(text);

🎮 Input

sf::Keyboard::WW key constant
sf::Keyboard::SpaceSpace key constant
sf::Keyboard::EscapeEscape key constant
sf::Mouse::getPosition()Global mouse position
sf::Mouse::getPosition(win)Mouse relative to window
event.key.codeKey that triggered event
event.mouseButtonMouse button info
if (sf::Keyboard::isKeyPressed(
    sf::Keyboard::W))
    player.y -= speed;
if (sf::Keyboard::isKeyPressed(
    sf::Keyboard::Left))
    player.x -= speed;

📐 Vectors & Math

# sf::Vector2f is the core type
sf::Vector2f pos(100.f, 200.f);
sf::Vector2f vel(0.f, 0.f);

# arithmetic
pos = pos + vel;
pos.x += vel.x * dt.asSeconds();

# transformable
sprite.setPosition(pos);
sprite.setRotation(45.f);
sprite.setScale(2.f, 2.f);

# common math
float len = std::sqrt(x*x + y*y);
float ang = std::atan2(dy, dx);

🎬 Animation (Sprite Sheet)

sf::Texture sheet;
sheet.loadFromFile("spritesheet.png");

int frameW = 32, frameH = 32;
int currentFrame = 0;
int totalFrames = 4;
float animSpeed = 0.15f;

sf::IntRect uvRect(0, 0,
    frameW, frameH);

# update each frame
timer += animSpeed;
if (timer >= 1.f) {
    timer = 0.f;
    currentFrame++;
    if (currentFrame >= totalFrames)
        currentFrame = 0;
}

# set the current frame region
uvRect.left = currentFrame * frameW;
sprite.setTextureRect(uvRect);
window.draw(sprite);

💥 Collision

# get bounding boxes
sf::FloatRect bounds1 =
    sprite1.getGlobalBounds();
sf::FloatRect bounds2 =
    sprite2.getGlobalBounds();

# AABB intersection
if (bounds1.intersects(bounds2)) {
    # collision!
}

# point inside rect
if (bounds1.contains(mousePos.x,
                       mousePos.y)) {
    # hover / click
}

# AABB parameters
bounds.left;   bounds.top;
bounds.width;  bounds.height;

⏱️ Time

sf::Clock clock;

# delta time
sf::Time dt = clock.restart();
float sec = dt.asSeconds();
int ms = dt.asMilliseconds();

# elapsed since last restart
sf::Time elapsed = clock.getElapsedTime();

# frame-rate independent movement
player.x += speed * dt.asSeconds();

# timer pattern
float cooldown = 0.f;
cooldown -= dt.asSeconds();
if (cooldown <= 0.f) {
    # fire, then reset
    cooldown = 1.f;
}

🛠️ Common Patterns

── FULL GAME LOOP ──
sf::RenderWindow window(sf::VideoMode(800,600), "Game");
window.setFramerateLimit(60);
sf::Clock clock;

while (window.isOpen()) {
    sf::Event e;
    while (window.pollEvent(e)) {
        if (e.type == sf::Event::Closed) window.close();
        if (e.type == sf::Event::KeyPressed
            && e.key.code == sf::Keyboard::Escape)
            window.close();
    }
    sf::Time dt = clock.restart();
    update(dt);
    window.clear(sf::Color::Black);
    draw(window);
    window.display();
}

── SPRITE ANIMATION ──
int frame = 0; float timer = 0.f;
int frameW = tex.getSize().x / cols;
int frameH = tex.getSize().y / rows;

timer += dt.asSeconds();
if (timer >= animSpeed) {
    timer = 0.f;
    frame = (frame + 1) % totalFrames;
}
sf::IntRect uv((frame % cols) * frameW,
              (frame / cols) * frameH,
              frameW, frameH);
sprite.setTextureRect(uv);

── COLLISION WITH RESOLUTION ──
sf::FloatRect a = player.getGlobalBounds();
sf::FloatRect b = wall.getGlobalBounds();
sf::FloatRect overlap;
if (a.intersects(b, overlap)) {
    # push out on smallest axis
    if (overlap.width < overlap.height) {
        # horizontal push
        player.x += (a.left < b.left)
            ? -overlap.width : overlap.width;
    } else {
        player.y += (a.top < b.top)
            ? -overlap.height : overlap.height;
    }
}

── SCROLLING BACKGROUND ──
float bgX = 0.f;
bgX -= speed * dt.asSeconds();
if (bgX <= -bgTex.getSize().x)
    bgX += bgTex.getSize().x;
bgSprite.setPosition(bgX, 0.f);
bgSprite2.setPosition(
    bgX + bgTex.getSize().x, 0.f);

── PARTICLE SYSTEM ──
struct Particle {
    sf::Vector2f pos, vel;
    float life;
};
std::vector<Particle> particles;

# spawn
particles.push_back({
    {x, y},
    {rand()%6-3, rand()%4-4},
    1.0f
});

# update
for (auto& p : particles) {
    p.pos += p.vel * dt.asSeconds();
    p.life -= dt.asSeconds();
}
particles.erase(
    std::remove_if(particles.begin(),
        particles.end(),
        [](auto& p){return p.life <= 0.f;}),
    particles.end());
