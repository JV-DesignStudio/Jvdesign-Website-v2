PICO-8 Lua Cheat Sheet

A one-page printable reference for the PICO-8 game loop, input, drawing, sound, memory and the most-used Lua patterns. Pin it next to your monitor!

🔄 Game Loop

function _init()
    0
end

function _update()
    if btn(0) then x -= 2 end
    if btn(1) then x += 2 end
end

function _draw()
    0)
    spr(1, x, y)
end

line(x1,y1,x2,y2,c)Draw a line

pset(x,y,c)Set a single pixel

🔊 Sound & Music

-- play sfx number n
sfx(0)

-- play music starting at pattern 0
music(0)

-- stop all music
music(-1)

-- set sfx playback speed
sfx(0, -1, 0, 8)

-- memory-mapped sfx registers
poke(0x3200, 1)  -- channel 0 volume

🧠 Memory & Map

peek(addr)Read a byte from memory address
poke(addr,val)Write a byte to memory address
mget(x,y)Get map tile at grid position
mset(x,y,v)Set map tile at grid position
sget(x,y)Get pixel from sprite sheet
sset(x,y,c)Set pixel on sprite sheet
memcpy(dst,src,len)Copy memory region
memset(addr,val,len)Fill memory region

Sprites: 0x0000-0x1FFF · Map: 0x2000-0x2FFF · Sprite flags: 0x3000-0x30FF

🛠️ Common Patterns

── SPRITE ANIMATION ──
frame = 0
function _update()
    frame += 0.25
    if frame >= 4 then frame = 0 end
end
function _draw()
    spr(1 + flr(frame), x, y)
end

── CAMERA FOLLOW ──
function _draw()
    cam_x = player.x - 64
    cam_y = player.y - 64
    camera(cam_x, cam_y)
    -- draw world here
end

── SIMPLE COLLISION ──
function collide(a, b)
    return a.x < b.x+8
       and a.x+8 > b.x
       and a.y < b.y+8
       and a.y+8 > b.y
end

── TILE COLLISION (MAP) ──
function solid(x, y)
    return mget(flr(x/8), flr(y/8)) > 0
end

── TIMER / COUNTDOWN ──
timer = 60
function _update()
    if timer > 0 then timer -= 1 end
end
