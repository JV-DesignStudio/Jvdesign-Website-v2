Jump Logic GDScript
if Input.is_action_just_pressed("ui_accept"):
    if is_on_floor():
        velocity.y = JUMP_VELOCITY

# Apply gravity when airborne
if not is_on_floor():

Coin Pickup GDScript
func _on_body_entered(body):
    if body.is_in_group("player"):
        GameState.score += 1
        emit_signal("coin_collected")
        queue_free()
